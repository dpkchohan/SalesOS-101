## 1. DATABASE SCHEMA (DETAILED)
=====================================

**Supabase PostgreSQL Tables:**

```sql
-- Organizations/Teams
CREATE TABLE organizations (
  id UUID PRIMARY KEY,
  name VARCHAR(255),
  website VARCHAR(255),
  created_at TIMESTAMP DEFAULT NOW()
);

-- Users
CREATE TABLE users (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  email VARCHAR(255) UNIQUE,
  name VARCHAR(255),
  role VARCHAR(50), -- 'admin', 'manager', 'sales'
  created_at TIMESTAMP DEFAULT NOW()
);

-- Companies
CREATE TABLE companies (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  name VARCHAR(255),
  website VARCHAR(255),
  industry VARCHAR(100),
  size VARCHAR(50), -- '1-10', '11-50', '51-200', etc.
  created_at TIMESTAMP DEFAULT NOW()
);

-- Contacts
CREATE TABLE contacts (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  company_id UUID REFERENCES companies(id),
  name VARCHAR(255),
  email VARCHAR(255),
  phone VARCHAR(20),
  title VARCHAR(100),
  tags JSONB, -- ["decision_maker", "technical"]
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Leads
CREATE TABLE leads (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  contact_id UUID REFERENCES contacts(id),
  assigned_to UUID REFERENCES users(id),
  stage VARCHAR(50), -- 'new', 'interested', 'qualified', 'negotiating', 'won', 'lost'
  value_usd INT, -- in paise/cents (store as integer)
  currency VARCHAR(10) DEFAULT 'USD',
  close_date DATE,
  source VARCHAR(100), -- 'cold_call', 'referral', 'inbound', etc.
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Interactions (Timeline)
CREATE TABLE interactions (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  lead_id UUID REFERENCES leads(id),
  created_by UUID REFERENCES users(id),
  type VARCHAR(50), -- 'call', 'email', 'meeting', 'note'
  date DATE,
  time TIME,
  gist TEXT, -- 1-2 line summary
  gains JSONB, -- ["budget_confirmed", "timeline_clear"]
  misses JSONB, -- ["needs_board_approval", "competitor_mentioned"]
  action_items JSONB, -- [{"title": "Send spec", "due_date": "2026-03-26"}]
  next_step TEXT,
  duration_minutes INT, -- for calls
  sentiment VARCHAR(50), -- 'positive', 'neutral', 'negative'
  created_at TIMESTAMP DEFAULT NOW()
);

-- Tasks
CREATE TABLE tasks (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  lead_id UUID REFERENCES leads(id),
  assigned_to UUID REFERENCES users(id),
  title VARCHAR(255),
  description TEXT,
  due_date DATE,
  completed_at TIMESTAMP,
  priority VARCHAR(20), -- 'high', 'medium', 'low'
  created_at TIMESTAMP DEFAULT NOW()
);

-- AI Signals (Red flags, coaching)
CREATE TABLE ai_signals (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  lead_id UUID REFERENCES leads(id),
  signal_type VARCHAR(50), -- 'deal_stalling', 'budget_unclear', 'timeline_vague'
  severity VARCHAR(20), -- 'high', 'medium', 'low'
  message TEXT,
  recommendation TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Proposals/Documents
CREATE TABLE documents (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  lead_id UUID REFERENCES leads(id),
  type VARCHAR(50), -- 'proposal', 'contract', 'case_study'
  title VARCHAR(255),
  content TEXT, -- HTML content
  pdf_url VARCHAR(500), -- Google Drive link
  format VARCHAR(50), -- 'pdf', 'googledocs', 'word'
  status VARCHAR(50), -- 'draft', 'sent', 'approved'
  sent_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Emails Sent
CREATE TABLE emails (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  lead_id UUID REFERENCES leads(id),
  to_email VARCHAR(255),
  subject VARCHAR(255),
  body TEXT,
  document_id UUID REFERENCES documents(id),
  sent_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW()
);
```

## 2. API ENDPOINTS (FULL SPEC)
===============================

**Base URL:** `/api/v1`

### **CONTACTS**
```
POST /contacts
Request: { name, email, phone, company_id, title, tags }
Response: { id, name, email, ... }

GET /contacts/:id
Response: { id, name, email, company, interactions[], tasks[] }

PUT /contacts/:id
Request: { name, email, phone, title }
Response: { id, ... updated fields }

DELETE /contacts/:id
Response: { success: true }

GET /contacts (with pagination)
Query: ?page=1&limit=20&search=john
Response: { data: [], total: 150, page: 1 }
```

### **LEADS**
```
POST /leads
Request: { contact_id, stage, value_usd, close_date }
Response: { id, contact_id, stage, ... }

GET /leads/:id
Response: { id, contact, stage, value, timeline[], tasks[] }

PUT /leads/:id
Request: { stage, value_usd, close_date, assigned_to }
Response: { id, ... }

GET /leads?stage=negotiating
Response: { data: [...], total: 15 }
```

### **INTERACTIONS**
```
POST /interactions
Request: { lead_id, type, gist, gains, misses, action_items, next_step, sentiment }
Response: { id, lead_id, type, ... }

GET /interactions/:lead_id
Response: { data: [ {date, gist, gains, misses, action_items} ], total: 20 }
```

### **TASKS**
```
POST /tasks
Request: { lead_id, title, due_date, assigned_to, priority }
Response: { id, title, due_date, ... }

PUT /tasks/:id
Request: { title, due_date, completed_at, priority }
Response: { id, ... }

GET /tasks?due_date=2026-03-26
Response: { data: [...], total: 5 }
```

### **AI SIGNALS (Red Flags)**
```
GET /ai-signals?lead_id=xyz
Response: { data: [ {signal_type, severity, message, recommendation} ] }
```

### **DOCUMENTS**
```
POST /documents
Request: { lead_id, type, title, content, format }
Response: { id, title, pdf_url, status: 'draft' }

PUT /documents/:id
Request: { status: 'sent', sent_at }
Response: { id, status, ... }
```

### **EMAILS**
```
POST /emails
Request: { lead_id, to_email, subject, body, document_id }
Response: { id, sent_at, ... }
```

### **Auth**
```
POST /auth/signup
Request: { email, password, org_name }
Response: { user: {}, token: 'jwt...' }

POST /auth/login
Request: { email, password }
Response: { user: {}, token: 'jwt...' }
```
